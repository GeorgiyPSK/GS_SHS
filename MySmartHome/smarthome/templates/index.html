      ⚡ Контекст запуска (ctx) 
      -- ctx заполняется при запуске по триггеру:
--   ctx.trigger   — "knx" | "mqtt" | "device_changed" | "timer"
--   ctx.script_id — id текущего скрипта
--   ctx.event     — полное событие
--   ctx.device_id, ctx.field, ctx.value — для device_changed
--   ctx.knx, ctx.mqtt, ctx.timer_id

log("Триггер: " .. tostring(ctx.trigger))

if ctx.trigger == "knx" and ctx.knx then
    log("KNX dest: " .. ctx.knx.dest)
end

if ctx.trigger == "device_changed" then
    log("Изменилось: " .. ctx.device_id .. "." .. ctx.field ..
        " = " .. tostring(ctx.value))
end
